class ModGrkDefiniteArticleService:
    def __init__(self):
        self.morphologyParser = MorphologyParser()

    def add_definite_article_to_lemma(self, lemma: str, pos: str, morph):
        # Gender
        gender = self.morphologyParser.parse(morph, "Gender")
        # Number
        number = self.morphologyParser.parse(morph, "Number")
        new_number = number if gender else 'Sing'
        # Case. Артикль леммы (а не токена) - это артикль всегда в именительном падеже
        case = 'Nom'

        new_morph = {"Gender": [gender], "Number": [new_number], "Case": [case]}
        return self.add_definite_article_to_token(lemma, pos, new_morph)

    def add_definite_article_to_token(self, token: str, pos: str, morph):
        articled_noun = '{0} {1}'
        vocative_noun = '{0}!'

        gender = self.morphologyParser.parse(morph, "Gender")
        number = self.morphologyParser.parse(morph, "Number")
        case = self.morphologyParser.parse(morph, "Case")

        if gender:
            if number and case:
                if case != CaseSpaCy.Voc.value:
                    definite_article = mod_grk_definite_articles_dict[gender][number][case]
                    result = articled_noun.format(definite_article, token)
                else:
                    result = vocative_noun.format(token)
        else:
            result = f"'Gender' field is empty for '{token}'"

        return result
