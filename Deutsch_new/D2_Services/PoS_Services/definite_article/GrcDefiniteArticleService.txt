class GrcDefiniteArticleService:
    def __init__(self):
        self.morphologyParser = MorphologyParser()

    def add_definite_article_to_lemma(self, lemma: str, pos: str, morph):
        # Искать артикль для леммы - это значит искать артикль всегда имея в виду только именительный падеж.
        # Род и число существительного остаются без изменения.
        gender = self.morphologyParser.parse(morph, "Gender")
        number = self.morphologyParser.parse(morph, "Number")

        # Если поле Gender не пустое, значит форма ед. ч. для слова существует и для поиска артикля именно для леммы
        # нужно передавать number='Sing'. Если же поле Gender - пустое, значит форма ед. ч. для слова не существует, и
        # слово употребляется только во мн.ч. В этом случае передаём тот Number, кот. пришёл от spaCy.
        new_number = number
        if gender:
            new_number = 'Sing'

        new_morph = {"Case": ["Nom"], "Gender": [gender], "Number": [new_number]}

        result = self.add_definite_article_to_token(lemma, pos, new_morph)
        return result

    def add_definite_article_to_token(self, token: str, pos: str, morph):

        articled_noun = '{0} {1}'
        vocative_noun = '{0}!'

        case = self.morphologyParser.parse(morph, "Case")
        number = self.morphologyParser.parse(morph, "Number")
        gender = self.morphologyParser.parse(morph, "Gender")

        # result = ''
        if case != CaseSpaCy.Voc.value:
            definite_article = grc_definite_articles_dict[gender][number][case]
            result = articled_noun.format(definite_article, token)
        else:  # если падеж - звательный!
            result = vocative_noun.format(token)

        return result

    # В gr-ru словарях артикль существительных указывают через запятую после самого существительного в им.п.
    # (а иногда после род. п. существительного), в любом случае артикль стоит на самом последнем месте.
    # Задача данного метода - переместить артикль на первое место, т. е. поставить его перед существительным, а заодно
    # проигнорировать род. п. существительного (если он указывался).
    def place_definite_article_on_the_first_place(self, input_text):
        results = list()
        words = [s for s in input_text.split(NEWLINE) if len(s)]  # взять в дальнейшую работу только непустые строки
        for word in words:
            # находим в словаре существительные (после них указан артикль)
            word_article_ending = ', {0}'
            masc_ending = word_article_ending.format(MASC_SG_NOM)
            fem_ending = word_article_ending.format(FEM_SG_NOM)
            neut_ending = word_article_ending.format(NEUT_SG_NOM)

            if word.endswith(masc_ending) or word.endswith(fem_ending) or word.endswith(neut_ending):
                pieces = word.split(', ')
                noun_nominative, definite_article = pieces[0], pieces[-1]
                res = definite_article + " " + noun_nominative
                results.append(res)
            else:
                results.append(word)
        output = NEWLINE.join(results)
        return output
