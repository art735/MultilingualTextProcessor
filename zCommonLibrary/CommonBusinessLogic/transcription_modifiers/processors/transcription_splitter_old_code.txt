import re


def process(word_or_sentence_transcription_or_text_containing_it, process_single_piece_cb):
    results = []

    # Если транскрипция содержит пробелы, т. е. это транскрипция целой фразы, разбиваем её на части
    # и работаем с каждой частью в отдельности, а потом в конце склеиваем обратно в одну строку.
    # Скобки в регулярном выражении (\s+) указывают на то, что разделители при разбивке строки нужно сохранить
    # (в т. ч. и по краям строки), т. к. эти пробелы являются значимыми в контексте абзацев ячейки .odt-файла.
    # РЕЗЮМЕ:
    # re.split(r'(\s+)', transcription) - разделители остаются
    # re.split(r'\s+', transcription) - разделители игнорируются
    pieces = re.split(r'(\s+)', word_or_sentence_transcription_or_text_containing_it)
    for piece in pieces:
        # Если piece имеет содержимое (т. е. не является пустой строкой) и это содержимое - не пробел(ы)
        if piece and not piece.isspace():
            # Данный флаг пригодится дважды:
            # 1) при стрипании '[' ДО обработки транскрипции
            # 2) при восстановлении '[' ПОСЛЕ обработки транскрипции
            is_left_bracket = piece.startswith('[')
            # Нет смысла стрипать и затем восстанавливать закрывающую квадратную скобку ']', т. к. она никак
            # не влияет на логику добавления аспирации или гортанной смычки. Там интересен только передний край
            # ударного слога.
            # is_right_bracket = piece.endswith(']')

            if is_left_bracket:
                piece = piece.lstrip('[')

            # Ключевой момент обработки кусочка транскрипции
            piece = process_single_piece_cb(piece)

            # Восстанавливаем '[', если она присутствовала в оригинальной транскрипции
            if is_left_bracket:
                piece = '[' + piece

        # добавляем КАЖДЫЙ piece в results независимо от того, обрабатывался он или нет
        results.append(piece)

    # Конкатенируем results через ПУСТУЮ СТРОКУ (а не через пробел), т. к. пробелы при разбивке строки сохранялись,
    # в т. ч. и по краям строки.
    output = ''.join(results)
    return output
